Bubble Sort with pass-by-pass visualizationvoid bubbleSort(int arr[], int n) {    cout << "\n--- Bubble Sort Processing ---\n";    for (int i = 0; i < n - 1; i++) {        for (int j = 0; j < n - i - 1; j++) {            if (arr[j] > arr[j + 1]) {                mySwap(arr[j], arr[j + 1]);            }        }        cout << "After Pass " << i + 1 << ": ";        printArray(arr, n);    }}// 2. Insertion Sort with element-by-element visualizationvoid insertionSort(int arr[], int n) {    cout << "\n--- Insertion Sort Processing ---\n";    for (int i = 1; i < n; i++) {        int key = arr[i];        int j = i - 1;        while (j >= 0 && arr[j] > key) {            arr[j + 1] = arr[j];            j = j - 1;        }        arr[j + 1] = key;        cout << "After inserting " << key << ": ";        printArray(arr, n);    }}// 3. Selection Sort with minimum selection visualizationvoid selectionSort(int arr[], int n) {    cout << "\n--- Selection Sort Processing ---\n";    for (int i = 0; i < n - 1; i++) {        int min_idx = i;        for (int j = i + 1; j < n; j++) {            if (arr[j] < arr[min_idx])                min_idx = j;        }        mySwap(arr[min_idx], arr[i]);        cout << "After Pass " << i + 1 << ": ";        printArray(arr, n);    }}// Merge Sort Helpersvoid merge(int arr[], int l, int m, int r, int totalSize) {    int n1 = m - l + 1;    int n2 = r - m;        // Create temp arrays    int *L = new int[n1];    int *R = new int[n2];        for (int i = 0; i < n1; i++) L[i] = arr[l + i];    for (int j = 0; j < n2; j++) R[j] = arr[m + 1 + j];        int i = 0, j = 0, k = l;    while (i < n1 && j < n2) {        if (L[i] <= R[j]) {            arr[k] = L[i];            i++;        } else {            arr[k] = R[j];            j++;        }        k++;    }        while (i < n1) {        arr[k] = L[i];        i++; k++;    }        while (j < n2) {        arr[k] = R[j];        j++; k++;    }        // cleanup memory    delete[] L;    delete[] R;        cout << "After Merge (indices " << l << " to " << r << "): ";    printArray(arr, totalSize);}// 4. Merge Sort (Recursive) with merge visualizationvoid mergeSort(int arr[], int l, int r, int totalSize) {    if (l < r) {        int m = l + (r - l) / 2;        mergeSort(arr, l, m, totalSize);        mergeSort(arr, m + 1, r, totalSize);        merge(arr, l, m, r, totalSize);    }}// Quick Sort Helperint partition(int arr[], int low, int high, int totalSize) {    int pivot = arr[high];    int i = (low - 1);        for (int j = low; j <= high - 1; j++) {        if (arr[j] < pivot) {            i++;            mySwap(arr[i], arr[j]);        }    }    mySwap(arr[i + 1], arr[high]);        cout << "After Partition (Pivot " << pivot << "): ";    printArray(arr, totalSize);        return (i + 1);}// 5. Quick Sort (Recursive) with partition visualizationvoid quickSort(int arr[], int low, int high, int totalSize) {    if (low < high) {        int pi = partition(arr, low, high, totalSize);        quickSort(arr, low, pi - 1, totalSize);        quickSort(arr, pi + 1, high, totalSize);    }}// Sorting Menu Systemvoid runSortingMenu() {    int size;    cout << "Enter the size of the array: ";    cin >> size;        int *originalArr = new int[size];    int *workingArr = new int[size];        cout << "Enter " << size << " integers:" << endl;    for (int i = 0; i < size; i++) {        cin >> originalArr[i];    }        int choice;    do {        // Reset working array to original state before every sort        copyArray(originalArr, workingArr, size);                cout << "\n--- SORTING MENU ---\n";        cout << "1. createBSTNode(int value) {    BSTNode* newNode = new BSTNode();    newNode->data = value;    newNode->left = NULL;    newNode->right = NULL;    return newNode;}// Insertion in BSTBSTNode* insertBST(BSTNode* root, int value) {    // Case 1: Empty tree    if (root == NULL) {        return createBSTNode(value);    }        // Case 2: Insert in left subtree    if (value < root->data) {        root->left = insertBST(root->left, value);    }    // Case 3: Insert in right subtree    else if (value > root->data) {        root->right = insertBST(root->right, value);    }    // Case 4: Duplicate (ignored)    return root;}// Search in BSTbool searchBST(BSTNode* root, int key) {    // Case 1: Empty tree or not found    if (root == NULL)        return false;        // Case 2: Found    if (root->data == key)        return true;        // Case 3: Search left    if (key < root->data)        return searchBST(root->left, key);        // Case 4: Search right    return searchBST(root->right, key);}// Find Minimum (for deletion)BSTNode* findMinBST(BSTNode* root) {    while (root->left != NULL)        root = root->left;    return root;}// Deletion in BST (handles all 3 cases)BSTNode* deleteBSTNode(BSTNode* root, int key) {    // Case 1: Tree empty    if (root == NULL)        return root;        // Traverse left    if (key < root->data) {        root->left = deleteBSTNode(root->left, key);    }    // Traverse right    else if (key > root->data) {        root->right = deleteBSTNode(root->right, key);    }    // Node found    else {        // Case 2: No child (leaf)        if (root->left == NULL && root->right == NULL) {            delete root;            return NULL;        }        // Case 3: One child (right)        else if (root->left == NULL) {            BSTNode* temp = root->right;            delete root;            return temp;        }        // Case 4: One child (left)        else if (root->right == NULL) {            BSTNode* temp = root->left;            delete root;            return temp;        }        // Case 5: Two children        BSTNode* temp = findMinBST(root->right); // inorder successor        root->data = temp->data;        root->right = deleteBSTNode(root->right, temp->data);    }    return root;}// Inorder Traversal (Left, Root, height balancing propertyFeatures: Insertion, deletion, searching, all rotations (LL, RR, LR, RL)Functions: Height balancing, rotations, traversals, node counting*/struct AVLNode {    int data;    AVLNode* left;    AVLNode* right;    int height;};// Create new AVL nodeAVLNode* createAVLNode(int value) {    AVLNode* node = new AVLNode();    node->data = value;    node->left = NULL;    node->right = NULL;    node->height = 1;  // New node has height 1 (leaf)    return node;}// Get height of AVL nodeint heightAVL(AVLNode* root) {    if (root == NULL)        return 0;    return root->height;}// Get balance factor of AVL nodeint getBalanceAVL(AVLNode* root) {    if (root == NULL)        return 0;    return heightAVL(root->left) - heightAVL(root->right);}// Right rotation (LL case)AVLNode* rightRotateAVL(AVLNode* y) {    AVLNode* x = y->left;    AVLNode* T2 = x->right;        x->right = y;    y->left = T2;        y->height = max(heightAVL(y->left), heightAVL(y->right)) + 1;    x->height = max(heightAVL(x->left), heightAVL(x->right)) + 1;        return x;}// Left rotation (RR case)AVLNode* leftRotateAVL(AVLNode* x) {    AVLNode* y = x->right;    AVLNode* T2 = y->left;        y->left = x;    x->right = T2;        x->height = max(heightAVL(x->left), heightAVL(x->right)) + 1;    y->height = max(heightAVL(y->left), heightAVL(y->right)) + 1;        return y;}// Insertion in AVL tree with rotationsAVLNode* insertAVL(AVLNode* root, int key) {    if (root == NULL)        return return root; // Duplicate values not allowed        root->height = 1 + max(heightAVL(root->left), heightAVL(root->right));        int balance = getBalanceAVL(root);        // LL Imbalance    if (balance > 1 && key < root->left->data)        return rightRotateAVL(root);        // RR Imbalance    if (balance < -1 && key > root->right->data)        return leftRotateAVL(root);        // && key < root->right->data) {        root->right = rightRotateAVL(root->right);        return leftRotateAVL(root);    }        return root;}// Find minimum node in AVL treeAVLNode* if (key < root->data)        root->left = deleteAVLNode(root->left, key);    else if (key > root->data)        root->right = deleteAVLNode(root->right, key);    else {        // Node with one or no child        if (root->left == NULL || root->right == NULL) {            AVLNode* temp = root->left ? root->left : root->right;                        if (temp == NULL) {                temp = return rightRotateAVL(root);    }        // RR Case    if (balance < -1 && getBalanceAVL(root->right) <= 0)        return leftRotateAVL(root);        // RL Case    if (balance < -1 && // Number of vertices    vector<vector<int>> adj; // Adjacency list    public:    // Constructor    Graph(int vertices) {        V = vertices;        adj.resize(V);    }        // Print Graph    void printGraph() {        cout << "Graph adjacency list:\n";        for (int i = 0; i < V; i++) {            cout << i << " -> ";            for (int j : adj[i])                cout << j << " ";            cout << endl;        }    }        // Add edge (undirected)    void addEdge(int u, int v) {        if (u >= V || v >= V) {            cout << "Invalid edge!\n";            return;        }        adj[u].push_back(v);        adj[v].push_back(u);    }        // Delete edge (undirected)    void deleteEdge(int u, int v) {        if (u >= V || v >= V) {            cout << "Invalid edge!\n";            return;        }        adj[u].erase(remove(adj[u].begin(), adj[u].end(), v), adj[u].end());        adj[v].erase(remove(adj[v].begin(), adj[v].end(), u), adj[v].end());    }        // Add new vertex    void addVertex() {        V++;        adj.resize(V);        cout << "Vertex " << V-1 << " added.\n";    }        // Delete vertex    void deleteVertex(int u) {        if (u >= V) {            cout << "Invalid vertex!\n";            return;        }                // Remove all edges to this vertex        for (int i = 0; i < V; i++) {            if (i == u) continue;                        adj[i].erase(remove(adj[i].begin(), adj[i].end(), u), adj[i].end());                        // Decrement vertex numbers higher than u            for (int &x : adj[i]) {                if (x > u) x--;            }        }                // Remove vertex u        adj.erase(adj.begin() + u);        V--;        cout << "Vertex " << u << " deleted.\n";                // Update adjacency lists for remaining vertices        for (int i = 0; i < V; i++) {            for (int &x : adj[i]) {                if (x > u) x--;            }        }    }};/* ================= MAIN FUNCTION ================= */int main() {    Graph g(5); // Create graph with 5 vertices (0 to 4)        // Add edges    g.addEdge(0, 1);    g.addEdge(0, 4);    g.addEdge(1, 2);    g.addEdge(1, 3);    g.addEdge(1, 4);    g.addEdge(2, 3);    g.addEdge(3, 4);        g.printGraph();        cout << "\nDelete edge (1,4):\n";    g.deleteEdge(1, 4);    g.printGraph();        cout << "\nAdd new vertex:\n";    g.addVertex();    g.printGraph();        cout << "\nAdd edge (5,0):\n";    g.addEdge(5, 0);    g.printGraph();        cout << "\nDelete vertex 2:\n";    g.deleteVertex(2);    g.printGraph();        return 0;}
